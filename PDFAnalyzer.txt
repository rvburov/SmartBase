    def get_base_designation(self, designation):
        """Извлекает базовый номер обозначения без кода документа и без суффикса"""
        if not designation:
            return ""
        
        # Убираем суффиксы -ЛУ и -УЛ
        clean_des = designation
        if clean_des.endswith('-ЛУ'):
            clean_des = clean_des[:-3]
        elif clean_des.endswith('-УЛ'):
            clean_des = clean_des[:-3]
        
        # Полный список всех кодов КД и ПД
        all_codes = []
        all_codes.extend(self.get_kd_name_mapping().keys())
        
        # Добавляем коды ПД
        for pattern in self.get_pd_name_mapping().keys():
            code = pattern.replace('XXXXX', '')
            if code and code not in all_codes:
                all_codes.append(code)
        
        # Убираем суффиксы из списка поиска
        all_codes = [c for c in all_codes if c not in ['-ЛУ', '-УЛ', 'ЛУ', 'УЛ']]
        all_codes = sorted(set(all_codes), key=len, reverse=True)
        
        # Ищем и удаляем код документа
        temp_des = clean_des
        for code in all_codes:
            if temp_des.endswith(f'-{code}'):
                temp_des = temp_des[:-len(f'-{code}')]
                break
            elif temp_des.endswith(code) and temp_des != code:
                temp_des = temp_des[:-len(code)]
                break
        
        # Теперь temp_des может содержать номер исполнения вида .024-01
        # Нужно убрать -01, оставив только .024
        # Ищем паттерн: точка, три цифры, дефис, цифры
        match = re.search(r'(\.[0-9]{3,4})-\d+$', temp_des)
        if match:
            # Убираем дефис и номер исполнения, оставляем только .024
            temp_des = temp_des[:match.start(1) + len(match.group(1))]
        
        # Убираем лишние дефисы в конце
        base = temp_des.rstrip('-')
        
        return base

    def get_document_full_name(self, designation, file_name="", all_designations=None):
        """
        Формирует полное наименование документа:
        - Для КД: [название изделия из спецификации] + [тип документа] + [суффикс]
        - Для ПД: [тип документа]
        
        all_designations: словарь всех обозначений и их названий из файлов
        """
        if not designation:
            return file_name if file_name else ""
        
        # Словарь соответствий
        kd_mapping = self.get_kd_name_mapping()
        
        # Получаем список всех исключений (удостоверяющих листов)
        exceptions = [code for code in kd_mapping.keys() if code.endswith('-УЛ') or code.endswith('-ЛУ')]
        exceptions.sort(key=len, reverse=True)
        
        # === СПЕЦИАЛЬНАЯ ОБРАБОТКА ДЛЯ ИСКЛЮЧЕНИЙ (УДОСТОВЕРЯЮЩИЕ ЛИСТЫ) ===
        # Проверяем, заканчивается ли обозначение на одно из исключений
        exception_code = None
        clean_for_exception = designation
        
        for exc in exceptions:
            if designation.endswith(exc):
                exception_code = exc
                # Убираем код исключения из обозначения, чтобы получить базовый номер
                clean_for_exception = designation[:-len(exc)]
                break
        
        if exception_code:
            doc_type = kd_mapping[exception_code]
            # Извлекаем базовый номер
            base_designation = self.get_base_designation(clean_for_exception)
            # Убираем номер исполнения
            if base_designation:
                base_designation = re.sub(r'-\d+$', '', base_designation)
            
            # Ищем название изделия
            product_name = ""
            if base_designation and base_designation in self.specification_names_cache:
                product_name = self.specification_names_cache[base_designation]
            elif base_designation and all_designations and base_designation in all_designations:
                product_name = all_designations[base_designation]
                self.specification_names_cache[base_designation] = product_name
            elif base_designation and file_name:
                # Пробуем извлечь из имени файла
                temp_name = file_name.replace('.pdf', '').strip()
                temp_name = temp_name.replace(designation, '').strip()
                temp_name = re.sub(r'[()\[\]_\-]+$', '', temp_name).strip()
                if temp_name:
                    product_name = temp_name
            
            # Формируем результат
            if product_name and product_name not in doc_type:
                return f"{product_name} {doc_type}".strip()
            return doc_type.strip()
        
        # === ОБЫЧНАЯ ОБРАБОТКА ДЛЯ ОСТАЛЬНЫХ ДОКУМЕНТОВ ===
        
        # Проверяем наличие суффиксов -ЛУ и -УЛ ТОЛЬКО В КОНЦЕ обозначения
        suffix_text = ""
        suffix_code = ""
        clean_designation = designation
        
        if designation.endswith('-ЛУ'):
            suffix_text = "Лист утверждения"
            suffix_code = "-ЛУ"
            clean_designation = designation[:-3]
        elif designation.endswith('-УЛ'):
            suffix_text = "Информационно-удостоверяющий лист"
            suffix_code = "-УЛ"
            clean_designation = designation[:-3]
        
        # Очищаем от номера исполнения для правильного поиска кода
        clean_designation = re.sub(r'-\d+(?=[A-ZА-Я])', '', clean_designation)
        
        # Определяем тип документа
        doc_type = ""
        base_designation = ""
        doc_code = None
        
        # Обычная обработка - извлекаем базовый номер и код документа
        base_designation, doc_code = self.extract_base_and_code(clean_designation)
        # Убираем номер исполнения из base_designation
        if base_designation:
            base_designation = re.sub(r'-\d+$', '', base_designation)
        doc_type = kd_mapping.get(doc_code, "")
        
        # Если код не найден, пробуем поискать в clean_designation
        if not doc_type and clean_designation:
            sorted_codes = sorted([c for c in kd_mapping.keys() if c not in ['-ЛУ', '-УЛ', 'ЛУ', 'УЛ']], 
                                key=len, reverse=True)
            for code in sorted_codes:
                if f'-{code}' in clean_designation:
                    doc_code = code
                    doc_type = kd_mapping.get(code, "")
                    break
                elif clean_designation.endswith(code) and len(code) < len(clean_designation):
                    doc_code = code
                    doc_type = kd_mapping.get(code, "")
                    break
        
        # Если тип документа не найден, пробуем ПД
        if not doc_type:
            pd_mapping = self.get_pd_name_mapping()
            for pattern, doc_type_pd in pd_mapping.items():
                pattern_regex = pattern.replace('XXXXX', r'\d{5}')
                if re.match(pattern_regex, designation) or re.match(pattern_regex, clean_designation):
                    doc_type = doc_type_pd
                    break
        
        # Пытаемся найти название изделия из спецификации
        product_name = ""
        
        # 1. Сначала ищем в кэше спецификаций
        if base_designation and base_designation in self.specification_names_cache:
            product_name = self.specification_names_cache[base_designation]
        # 2. Пробуем искать в all_designations по base_designation
        elif base_designation and all_designations and base_designation in all_designations:
            product_name = all_designations[base_designation]
            self.specification_names_cache[base_designation] = product_name
        # 3. Пробуем искать по полному обозначению
        elif clean_designation and all_designations and clean_designation in all_designations:
            product_name = all_designations[clean_designation]
            if base_designation:
                self.specification_names_cache[base_designation] = product_name
        # 4. Пробуем искать исходное обозначение
        elif designation and all_designations and designation in all_designations:
            product_name = all_designations[designation]
            if base_designation:
                self.specification_names_cache[base_designation] = product_name
        
        # 5. Если все еще нет названия, используем имя файла
        if not product_name and file_name:
            base_name = file_name.replace('.pdf', '').strip()
            for des in [designation, clean_designation]:
                if des and des in base_name:
                    base_name = base_name.replace(des, '').strip()
            base_name = base_name.replace('-ЛУ', '').replace('-УЛ', '').strip()
            base_name = re.sub(r'[()\[\]_\-]+$', '', base_name).strip()
            base_name = re.sub(r'-\d+', '', base_name).strip()
            if base_name:
                product_name = base_name
        
        # Формируем результат
        result = ""
        
        if doc_type:
            if product_name and product_name not in doc_type:
                result = f"{product_name} {doc_type}"
            else:
                result = doc_type
        elif product_name:
            result = product_name
        else:
            result = file_name.replace('.pdf', '') if file_name else ""
        
        # Добавляем суффикс, если он есть и еще не добавлен
        if suffix_text and suffix_text not in result:
            if suffix_text not in doc_type:
                result = f"{result} {suffix_text}"
        
        return result.strip()

    def extract_base_and_code(self, designation):
        """Извлекает базовый номер и код документа из обозначения"""
        if not designation:
            return designation, None
        
        kd_mapping = self.get_kd_name_mapping()
        all_codes = [c for c in kd_mapping.keys() if c not in ['-ЛУ', '-УЛ', 'ЛУ', 'УЛ']]
        all_codes.sort(key=len, reverse=True)
        
        # Ищем код документа в конце строки
        found_code = None
        base = designation
        
        for code in all_codes:
            if designation.endswith(code):
                found_code = code
                base = designation[:-len(code)]
                break
            # Проверяем с дефисом
            if designation.endswith(f'-{code}'):
                found_code = code
                base = designation[:-len(f'-{code}')]
                break
        
        # Если код не найден, ищем более сложные паттерны
        if not found_code:
            # Формат: АААА.123456.123 или АААА.123456.123-14 (где 14 - это номер исполнения, а не код)
            # Сначала ищем базовый шаблон с возможным номером исполнения
            match = re.match(r'([A-ZА-Я]{4}\.\d{6}\.\d{3,4}(?:-\d+)?)', designation)
            if match:
                base = match.group(1)
                # Остаток после базового номера - это код документа
                remainder = designation[len(base):].lstrip('-')
                if remainder and remainder in all_codes:
                    found_code = remainder
                elif remainder:
                    # Проверяем, не начинается ли остаток с какого-то кода
                    for code in all_codes:
                        if remainder.startswith(code):
                            found_code = code
                            # Корректируем base, убирая лишнее
                            base = designation[:-len(remainder)]
                            break
        
        # Очищаем base от лишних дефисов в конце
        base = base.rstrip('-')
        
        return base, found_code

    def get_code_from_designation(self, designation):
        """Извлекает код документа из обозначения КД"""
        if not designation:
            return None
        
        # Сначала проверяем полные исключения (например Э3-УЛ)
        kd_order = self.get_kd_order()
        
        if designation in kd_order:
            return designation
        
        # Проверяем без суффикса
        clean_designation = designation
        suffix = ""
        
        if designation.endswith('-ЛУ'):
            clean_designation = designation[:-3]
            suffix = '-ЛУ'
        elif designation.endswith('-УЛ'):
            clean_designation = designation[:-3]
            suffix = '-УЛ'
        
        # Ищем код в очищенном обозначении
        codes = sorted([c for c in kd_order.keys() if c not in ['-ЛУ', '-УЛ']], key=len, reverse=True)
        
        for code in codes:
            if f'-{code}' in clean_designation:
                # Возвращаем код с суффиксом, если он был
                return f"{code}{suffix}" if suffix else code
            if clean_designation.endswith(code) and len(code) < len(clean_designation):
                return f"{code}{suffix}" if suffix else code
        
        # Спецификации
        return 'СПЕЦИФИКАЦИЯ'

    def find_pdf_files(self, folder_path, recursive=True):
        """Рекурсивно находит все PDF файлы в папке и подпапках"""
        pdf_files = []
        
        if recursive:
            for root, dirs, files in os.walk(folder_path):
                for file_name in files:
                    if file_name.lower().endswith('.pdf'):
                        full_path = os.path.join(root, file_name)
                        relative_path = os.path.relpath(root, folder_path)
                        if relative_path == '.':
                            relative_path = ''
                        pdf_files.append({
                            'full_path': full_path,
                            'file_name': file_name,
                            'folder': relative_path,
                            'mtime': os.path.getmtime(full_path)
                        })
        else:
            for file_name in os.listdir(folder_path):
                if file_name.lower().endswith('.pdf'):
                    full_path = os.path.join(folder_path, file_name)
                    pdf_files.append({
                        'full_path': full_path,
                        'file_name': file_name,
                        'folder': '',
                        'mtime': os.path.getmtime(full_path)
                    })
        
        return pdf_files

            'МД-УЛ', 'ЭМД-УЛ', 'МС-УЛ', 'Д23-УЛ', 'НДЗ-УЛ', 'НД-УЛ', 'ТУ-ЛУ', 'ВЭ-ЛУ', 'ПС-ЛУ', 'ЭСБ-УЛ'
        ]
        
        # Удаляем дубликаты и сортируем
        kd_codes = sorted(set(kd_codes), key=len, reverse=True)
        kd_pattern = '|'.join(re.escape(code) for code in kd_codes)
        
        # Паттерн с дефисом - сохраняем суффиксы
        pattern_with_hyphen = rf'''
            \b
            [A-ZА-Я]{{4}}\.\d{{6}}\.\d{{3,4}}
            (?:-\d{{2}}\.\d{{2}})?
            (?:-\d+)?
            (?:-{kd_pattern})?
            (?:-ЛУ)?   # Сохраняем суффикс -ЛУ
            (?:-УЛ)?   # Сохраняем суффикс -УЛ
            \b
        '''
        
        compiled_with = re.compile(pattern_with_hyphen, re.VERBOSE | re.IGNORECASE)
        match = compiled_with.search(text)
        
        if match:
            result = match.group(0)
            result = re.sub(r'\s+', ' ', result)
            return result.strip()
        
        # Паттерн слитный
        pattern_without_hyphen = rf'''
            \b
            [A-ZА-Я]{{4}}\.\d{{6}}\.\d{{3,4}}
            (?:-\d{{2}}\.\d{{2}})?
            (?:-\d+)?
            (?:{kd_pattern})?
            (?:-ЛУ)?
            (?:-УЛ)?
            (?!\d)
            \b
        '''
        
        compiled_without = re.compile(pattern_without_hyphen, re.VERBOSE | re.IGNORECASE)
        match = compiled_without.search(text)
        
        if match:
            result = match.group(0)
            result = re.sub(r'\s+', ' ', result)
            return result.strip()
        
        return ""


    def extract_numbers_PD(self, text):
        """Извлекает номер ПД (программной документации) по ГОСТ 19.101"""
        
        # КОДЫ ДОКУМЕНТОВ ЕСПД (ГОСТ 19.101)
        pd_codes = [
            'ТЗ', 'ПЗ', 'ПО', 'ПР', 'СП', 'ТК', 'РО', 'РП', 'РС', 'РТ', 'РЭ',
            'ИО', 'ИП', 'ИС', 'ИТ', 'ВЭ', 'ВД', 'ВМ', 'ВЗ', 'ЗИ', 'ОП', 'ОО', 
            'ОА', 'ОС', 'ОД', 'ЛУ', 'ЛС', 'АИ', 'АТ', 'АВ', 'РБ', 'ИБ', 'МП', 
            'БД', 'АП', 'ТМ', 'ТЗ-ЛУ', 'ПЗ-ЛУ', 'ПО-ЛУ', 'ПР-ЛУ', 'РО-ЛУ', 
            'РП-ЛУ', 'ОП-ЛУ', 'ВЭ-ЛУ', 'НД', 'НДЗ', 'НД-УЛ', 'НДЗ-УЛ',
        ]
        
        pd_codes.sort(key=len, reverse=True)
        pd_pattern = '|'.join(pd_codes)
        
        # Паттерн 1: ПД с цифровым кодом (АБВГ.01401-01 12 01-ЛУ)
        pattern_numeric = rf'''
        \b
        (
        [A-ZА-Я]{{4}}\.\d{{5}}
        -\d{{2}}
        \s+\d{{2}}
        \s+\d{{2}}
        (?:-\d)?
        (?:-ЛУ)?
        )
        \b
        '''
        
        # Паттерн 2: ПД с буквенным кодом (АБВГ.01401-01 ТЗ 01-ЛУ)
        pattern_letter = rf'''
        \b
        (
        [A-ZА-Я]{{4}}\.\d{{5}}
        -\d{{2}}
        \s+(?:{pd_pattern})
        \s+\d{{2}}
        (?:-\d)?
        (?:-ЛУ)?
        )
        \b
        '''
        
        # Паттерн 3: ПД без кода (АБВГ.01401-01 или АБВГ.01401-01-ЛУ)
        pattern_simple = rf'''
        \b
        (
        [A-ZА-Я]{{4}}\.\d{{5}}
        -\d{{2}}
        (?:-ЛУ)?
        )
        \b
        '''
        
        # Паттерн 4: ПД с диском (АБВГ.01401-01НДЗ или АБВГ.01401-01НДЗ-УЛ)
        pattern_disk = rf'''
        \b
        (
        [A-ZА-Я]{{4}}\.\d{{5}}
        -\d{{2}}
        (?:{pd_pattern})
        (?:-УЛ)?
        )
        \b
        '''
        
        # Паттерн 5: Слитный код без дефиса (АБВГ.01401-01ТЗ 01)
        pattern_combined = rf'''
        \b
        (
        [A-ZА-Я]{{4}}\.\d{{5}}
        -\d{{2}}
        (?:{pd_pattern})
        \s+\d{{2}}
        (?:-\d)?
        (?:-ЛУ)?
        )
        \b
        '''
        
        for pattern in [pattern_numeric, pattern_letter, pattern_simple, pattern_disk, pattern_combined]:
            compiled = re.compile(pattern, re.VERBOSE | re.IGNORECASE)
            match = compiled.search(text)
            if match:
                result = match.group(1)
                result = re.sub(r'\s+', ' ', result)
                return result.strip()
        
        return ""
    
    def extract_numbers_TD(self, text):
        """Извлекает номер ТД по строгому формату АААА.00000.00000, удаляя приставки"""
        # Паттерн для ТД: 4 буквы, точка, 5 цифр, точка, 5 цифр, опциональные приставки
        pattern = r'([A-ZА-Я]{4}\.\d{5}\.\d{5})'
        match = re.search(pattern, text)
        
        if match:
            # Возвращаем только чистый номер без приставок
            return match.group(1)
        else:
            # Попробуем найти номер с приставками и удалить их
            # Этот паттерн ищет номер с возможными приставками после него
            pattern_with_suffix = r'([A-ZА-Я]{4}\.\d{5}\.\d{5})(?:[A-ZА-Я]+)?'
            match_with_suffix = re.search(pattern_with_suffix, text)
            return match_with_suffix.group(1) if match_with_suffix else ""
    
    def parse_filename(self, filename):
        """Парсит имя файла для извлечения номеров КД, ПД и ТД"""
        base_name = filename[:-4]  # Убираем .pdf
        
        # Ищем номер КД в имени файла
        numbers_KD = self.extract_numbers_KD(base_name)
        
        # Ищем номер ПД в имени файла (программная документация)
        numbers_PD = self.extract_numbers_PD(base_name)
        
        # Ищем номер ТД в имени файла (технологическая документация)
        numbers_TD = self.extract_numbers_TD(base_name)
        
        # ВАЖНО: numbers_KD должен содержать ПОЛНОЕ обозначение (включая -ЛУ, -УЛ)
        # для этого нужно изменить extract_numbers_KD, чтобы он не обрезал суффиксы
        
        # Извлекаем название документа из имени файла
        name = base_name
        
        # Удаляем номера из названия
        if numbers_TD:
            name = name.replace(numbers_TD, "").strip()
        if numbers_PD:
            name = name.replace(numbers_PD, "").strip()
        if numbers_KD:
            name = name.replace(numbers_KD, "").strip()
        
        # Убираем суффиксы из названия
        name = name.replace('-ЛУ', '').replace('-УЛ', '').strip()
        
        # Убираем скобки и лишние пробелы
        name = re.sub(r'[()\[\]]', '', name).strip()
        name = ' '.join(name.split())
        
        return numbers_KD, numbers_PD, numbers_TD, name
    
    def calculate_md5(self, filepath):
        """Вычисляет MD5 хеш-сумму файла"""
        hash_md5 = hashlib.md5()
        try:
            with open(filepath, "rb") as f:
                for chunk in iter(lambda: f.read(4096), b""):
                    hash_md5.update(chunk)
            return hash_md5.hexdigest().upper()
        except Exception as e:
            return "Ошибка"
    
    # ИЗМЕНЕННЫЙ МЕТОД: Определение формата листа по размерам с погрешностью 30 мм
    def determine_page_format(self, width, height):
        """Определяет формат листа по размерам в мм с погрешностью 30 мм"""
        # Приводим размеры к стандартному порядку (меньшая сторона, большая сторона)
        w, h = min(width, height), max(width, height)
                
        # Основные форматы по ГОСТ (размеры в мм)
        formats = {
            'A0': (841, 1189),
            'A1': (594, 841),
            'A2': (420, 594),
            'A3': (297, 420),
            'A4': (210, 297),
            'A5': (148, 210),
        }
        
        # Дополнительные форматы (производные)
        additional_formats = {
            'A0×2': (1189, 1682),
            'A0×3': (1189, 2523),
            'A1×3': (841, 1783),
            'A1×4': (841, 2378),
            'A2×3': (594, 1261),
            'A2×4': (594, 1682),
            'A2×5': (594, 2102),
            'A3×3': (420, 891),
            'A3×4': (420, 1189),
            'A3×5': (420, 1486),
            'A3×6': (420, 1783),
            'A3×7': (420, 2080),
            'A4×3': (297, 630),
            'A4×4': (297, 841),
            'A4×5': (297, 1051),
            'A4×6': (297, 1261),
            'A4×7': (297, 1471),
            'A4×8': (297, 1682),
            'A4×9': (297, 1982),
        }
        
        # Проверяем основные форматы с погрешностью 30 мм
        tolerance = 30.0
        for fmt, (std_w, std_h) in formats.items():
            if (abs(w - std_w) <= tolerance and abs(h - std_h) <= tolerance) or \
               (abs(w - std_h) <= tolerance and abs(h - std_w) <= tolerance):
                return fmt, None
        
        # Проверяем дополнительные форматы с погрешностью 30 мм
        for fmt, (std_w, std_h) in additional_formats.items():
            if (abs(w - std_w) <= tolerance and abs(h - std_h) <= tolerance) or \
               (abs(w - std_h) <= tolerance and abs(h - std_w) <= tolerance):
                return 'Доп. форматы', fmt
        
        # Если формат не определен ни по одному из стандартов, возвращаем точный размер
        return 'Доп. форматы', f"{int(w)}×{int(h)}"
    
    # ИЗМЕНЕННЫЙ МЕТОД: Подсчет форматов листов в PDF
    def count_page_formats(self, pdf_path):
        """Подсчитывает количество листов каждого формата в PDF документе"""
        # Инициализируем счетчики для всех форматов
        format_counts = {
            'A4': 0,
            'A3': 0,
            'A2': 0,
            'A1': 0,
            'A0': 0,
            'Доп. форматы': 0
        }
        
        # Словарь для хранения нестандартных размеров и их количества
        non_standard_sizes = {}
        
        try:
            with open(pdf_path, 'rb') as file:
                pdf = PyPDF2.PdfReader(file)
                
                for page_num in range(len(pdf.pages)):
                    page = pdf.pages[page_num]
                    
                    # Получаем размер страницы в пунктах (1 пункт = 1/72 дюйма)
                    # Конвертируем в миллиметры (1 дюйм = 25.4 мм)
                    media_box = page.mediabox
                    
                    # Получаем размеры в пунктах и конвертируем в мм
                    width_pts = float(media_box.width)
                    height_pts = float(media_box.height)
                    
                    # Конвертация в мм
                    width_mm = width_pts * 25.4 / 72
                    height_mm = height_pts * 25.4 / 72
                    
                    # Округляем до целых
                    width_mm = round(width_mm)
                    height_mm = round(height_mm)
                    
                    
                    # Определяем формат страницы и получаем нестандартный размер если есть
                    page_format, non_standard_size = self.determine_page_format(width_mm, height_mm)
                                        
                    # Увеличиваем соответствующий счетчик
                    if page_format in format_counts:
                        format_counts[page_format] += 1
                    
                    # Если есть нестандартный размер, записываем его
                    if non_standard_size:
                        size_key = non_standard_size
                        if size_key in non_standard_sizes:
                            non_standard_sizes[size_key] += 1
                        else:
                            non_standard_sizes[size_key] = 1
                
        except Exception as e:
            print(f"Ошибка при подсчете форматов для {pdf_path}: {e}")
            # В случае ошибки возвращаем пустые значения
        
        # Формируем строку с нестандартными размерами
        non_standard_str = ""
        if non_standard_sizes:
            size_strings = []
            for size, count in non_standard_sizes.items():
                if count > 1:
                    size_strings.append(f"{size} ({count} шт.)")
                else:
                    size_strings.append(size)
            non_standard_str = ", ".join(size_strings)
        
        return format_counts, non_standard_str
    
    def format_excel_file(self, excel_file_path):
        """Применяет форматирование к Excel файлу: настройка ширины столбцов, шрифт, выравнивание, границы, высота строк"""
        try:
            from openpyxl.styles import Alignment, Font, Border, Side
            from openpyxl.utils import get_column_letter
            
            # Загружаем рабочую книгу
            wb = load_workbook(excel_file_path)
            ws = wb.active
            
            # Настройки шрифта
            font = Font(name='Times New Roman', size=12)
            
            # Настройки выравнивания
            center_alignment = Alignment(
                horizontal='center',
                vertical='center',
                wrap_text=True  # Перенос текста
            )
            
            # Настройки границ
            thin_border = Border(
                left=Side(style='thin'),
                right=Side(style='thin'),
                top=Side(style='thin'),
                bottom=Side(style='thin')
            )

            # Определяем диапазон таблицы (все ячейки с данными)
            max_row = ws.max_row
            max_col = ws.max_column
            
            # Применяем форматирование ко всем ячейкам в диапазоне таблицы
            for row in range(1, max_row + 1):
                for col in range(1, max_col + 1):
                    cell = ws.cell(row=row, column=col)
                    cell.font = font
                    cell.alignment = center_alignment
                    cell.border = thin_border
            
            # Настраиваем ширину для каждого столбца
            for column in ws.columns:
                max_length = 0
                column_letter = get_column_letter(column[0].column)
                
                # Находим максимальную длину текста в столбце
                for cell in column:
                    try:
                        if cell.value:
                            cell_length = len(str(cell.value))
                            max_length = max(max_length, cell_length)
                    except:
                        pass
                
                # Устанавливаем ширину столбца в зависимости от названия
                column_name = ws.cell(row=1, column=column[0].column).value
                
                if column_name == '№ п/п':
                    adjusted_width = 6
                elif column_name == 'Наименование документа':
                    adjusted_width = 27
                elif column_name == 'Формат файла':
                    adjusted_width = 10
                elif column_name == 'Контрольная сумма':
                    adjusted_width = 44
                elif column_name == 'Объем, МБ':
                    adjusted_width = 9
                elif column_name == 'Физических листов':
                    adjusted_width = 14
                elif column_name == 'Обозначение ЭН':
                    adjusted_width = 19
                elif column_name == 'Гриф секретности':
                    adjusted_width = 14
                elif column_name == 'Примечание':
                    adjusted_width = 14
                elif column_name in ['Обозначение файла КД', 'Обозначение файла ТД']:
                    adjusted_width = 27
                elif column_name in ['Обозначение документа ТД', 'Обозначение документа КД']:
                    adjusted_width = 29
                elif column_name in ['А4', 'А3', 'А2', 'А1', 'А0']:
                    adjusted_width = 4
                elif column_name == 'Доп. форматы':
                    adjusted_width = 11
                elif column_name == 'Листов формата А4':
                    adjusted_width = 12
                else:
                    adjusted_width = min(max_length + 2, 50)
                
                ws.column_dimensions[column_letter].width = adjusted_width
            
            # Устанавливаем высоту первой строки (заголовок) - 60
            ws.row_dimensions[1].height = 60
            
            # Устанавливаем высоту остальных строк - 30
            for row in range(2, ws.max_row + 1):
                ws.row_dimensions[row].height = 30
            
            # Сохраняем изменения
            wb.save(excel_file_path)
            
        except Exception as e:
            print(f"Ошибка при форматировании Excel файла: {e}")

    def update_progress(self, current, total, current_file=""):
        """Обновляет прогресс обработки"""
        if hasattr(self, 'progress_dialog') and self.progress_dialog:
            percentage = int((current / total) * 100)
            self.progress_bar.setValue(current)
            
            if current_file:
                # Обрезаем длинное имя файла для отображения
                display_file = current_file
                if len(current_file) > 40:
                    display_file = "..." + current_file[-37:]
                
                self.progress_label.setText(
                    f'Обработка файлов: {current} из {total} ({percentage}%)\n'
                    f'Текущий файл: {display_file}'
                )
            else:
                self.progress_label.setText(
                    f'Обработка файлов: {current} из {total} ({percentage}%)'
                )
            
            # Принудительная обработка событий для кнопок
            QApplication.processEvents()
            
            # Возвращаем True если обработка продолжается, False если отменена
            return not self.processing_cancelled
        
        return True

    def analyze_pdf(self):
        folder_path = self.pdf_path_input.text()
        save_path = self.save_path_input.text()
        recursive = self.recursive_checkbox.isChecked()
        
        if not folder_path:
            self.show_message_box('Ошибка', 'Укажите путь к папке с PDF файлами', QMessageBox.Warning)
            return
            
        if not os.path.exists(folder_path):
            self.show_message_box('Ошибка', f'Папка {folder_path} не существует', QMessageBox.Warning)
            return
        
        if not save_path:
            save_path = os.path.join(folder_path, 'pdf_info.xlsx')
        elif os.path.isdir(save_path):
            save_path = os.path.join(save_path, 'pdf_info.xlsx')
        
        try:
            # Отключаем кнопку на время обработки
            self.start_button.setEnabled(False)
            self.start_button.setText('Поиск файлов...')
            QApplication.processEvents()
            
            # Находим все PDF файлы (первый проход - подсчет)
            self.start_button.setText('Подсчет файлов...')
            QApplication.processEvents()
            
            pdf_files = self.find_pdf_files(folder_path, recursive)
            
            if not pdf_files:
                self.show_message_box('Ошибка', 'В указанной папке не найдено PDF файлов', QMessageBox.Warning)
                self.start_button.setEnabled(True)
                self.start_button.setText('Запустить анализ')
                return
            
            total_files = len(pdf_files)
            
            # Показываем диалоговое окно прогресса
            progress_dialog = self.show_progress_dialog(total_files)
            
            # Обрабатываем каждый найденный файл
            processed_files = 0
            numbers_KD_list = []
            numbers_TD_list = []
            numbers_PD_list = []
            names = []
            page_counts = []
            file_names = []
            file_paths = []
            file_mtimes = []
            insurance_KD = []
            insurance_TD = []
            file_sizes = []
            file_checksums = []
            folders = []
            error_files = []
            number_pp = []
            file_formats = []
            en_designations = []
            secrecy_grades = []
            notes = []

            # Списки для форматов
            a4_counts = []
            a3_counts = []
            a2_counts = []
            a1_counts = []
            a0_counts = []
            additional_counts = []
            a4_sheets_counts = []
            
            # Обрабатываем каждый найденный файл
            for idx, file_info in enumerate(pdf_files, 1):
                file_path = file_info['full_path']
                file_name = file_info['file_name']
                folder_name = file_info['folder']
                
                # Проверяем, не была ли отменена обработка
                if self.processing_cancelled:
                    # Закрываем диалог прогресса
                    self.close_progress_dialog()
                    
                    # Показываем центрированное сообщение об отмене
                    self.show_message_box('Отменено', 'Обработка файлов была отменена пользователем.', QMessageBox.Information)
                    
                    self.start_button.setEnabled(True)
                    self.start_button.setText('Запустить анализ')
                    return
                
                try:
                    # Обновляем прогресс
                    if not self.update_progress(idx, total_files, file_name):
                        # Если update_progress вернул False (отмена)
                        self.close_progress_dialog()
                        self.show_message_box('Отменено', 'Обработка файлов была отменена пользователем.', QMessageBox.Information)
                        self.start_button.setEnabled(True)
                        self.start_button.setText('Запустить анализ')
                        return
                    
                    # Получаем размер файла
                    file_size = os.path.getsize(file_path) / (1024 * 1024)
                    file_size = round(file_size, 2)
                    
                    # Вычисляем MD5 хеш-сумму файла
                    file_checksum = self.calculate_md5(file_path)
                    
                    # Парсим имя файла
                    numbers_KD, numbers_PD, numbers_TD, name = self.parse_filename(file_name)

                    # В блоке try (успешная обработка)
                    with open(file_path, 'rb') as file:
                        pdf = PyPDF2.PdfReader(file)
                        page_count = len(pdf.pages)
                        
                        # Подсчет форматов
                        format_counts, non_standard_str = self.count_page_formats(file_path)
                        
                        # ДОБАВЛЯЕМ КАЖДЫЙ ЭЛЕМЕНТ ТОЛЬКО ОДИН РАЗ
                        numbers_KD_list.append(numbers_KD)
                        numbers_TD_list.append(numbers_TD)
                        numbers_PD_list.append(numbers_PD)
                        names.append(name)
                        page_counts.append(page_count)
                        file_names.append(file_name)
                        file_paths.append(file_path)
                        file_mtimes.append(file_info.get('mtime', 0))
                        folders.append(folder_name)
                        insurance_KD.append(f"{numbers_KD}.pdf" if numbers_KD else "")
                        insurance_TD.append(f"{numbers_TD}.pdf" if numbers_TD else "")
                        file_sizes.append(file_size)
                        file_checksums.append(file_checksum)
                        file_formats.append("pdf")
                        en_designations.append("")
                        secrecy_grades.append("н/с")
                        notes.append("")
                        a4_counts.append(format_counts['A4'] if format_counts['A4'] > 0 else "")
                        a3_counts.append(format_counts['A3'] if format_counts['A3'] > 0 else "")
                        a2_counts.append(format_counts['A2'] if format_counts['A2'] > 0 else "")
                        a1_counts.append(format_counts['A1'] if format_counts['A1'] > 0 else "")
                        a0_counts.append(format_counts['A0'] if format_counts['A0'] > 0 else "")
                        
                        # Обработка дополнительных форматов
                        if non_standard_str:
                            if '×' in non_standard_str and not non_standard_str.startswith('A'):
                                additional_counts.append(non_standard_str)
                            else:
                                additional_pages = format_counts['Доп. форматы']
                                additional_counts.append(additional_pages if additional_pages > 0 else "")
                        elif format_counts['Доп. форматы'] > 0:
                            additional_counts.append(format_counts['Доп. форматы'])
                        else:
                            additional_counts.append("")
                        
                        # Расчет листов А4
                        a4_sheets = 0
                        format_to_multiplier = {
                            'A4×3': 3, 'A4×4': 4, 'A4×5': 5, 'A4×6': 6, 'A4×7': 7, 'A4×8': 8, 'A4×9': 9,
                            'A3×3': 6, 'A3×4': 8, 'A3×5': 10, 'A3×6': 12, 'A3×7': 14,
                            'A2×3': 12, 'A2×4': 16, 'A2×5': 20,
                            'A1×3': 24, 'A1×4': 32,
                            'A0×2': 32, 'A0×3': 48,
                        }
                        
                        if format_counts['A4'] > 0:
                            a4_sheets += format_counts['A4']
                        if format_counts['A3'] > 0:
                            a4_sheets += format_counts['A3'] * 2
                        if format_counts['A2'] > 0:
                            a4_sheets += format_counts['A2'] * 4
                        if format_counts['A1'] > 0:
                            a4_sheets += format_counts['A1'] * 8
                        if format_counts['A0'] > 0:
                            a4_sheets += format_counts['A0'] * 16
                        
                        if format_counts['Доп. форматы'] > 0 and non_standard_str:
                            parts = non_standard_str.split(', ')
                            for part in parts:
                                count = 1
                                format_name = part
                                if '(' in part and 'шт.' in part:
                                    try:
                                        count_str = part.split('(')[1].split()[0]
                                        count = int(count_str)
                                        format_name = part.split('(')[0].strip()
                                    except:
                                        pass
                                if format_name in format_to_multiplier:
                                    a4_sheets += count * format_to_multiplier[format_name]
                        
                        a4_sheets_counts.append(a4_sheets if a4_sheets > 0 else "")

                    processed_files += 1
                    number_pp.append(processed_files)
                    
                except Exception as e:
                    error_files.append(f"{file_name}: {str(e)}")
                    # В случае ошибки добавляем пустые значения для ВСЕХ списков
                    numbers_KD_list.append("")
                    numbers_TD_list.append("")
                    numbers_PD_list.append("")
                    names.append("")
                    page_counts.append(0)
                    file_names.append(file_name)
                    file_paths.append(file_path)
                    file_mtimes.append(file_info.get('mtime', 0))
                    folders.append(folder_name)
                    insurance_KD.append("")
                    insurance_TD.append("")
                    file_sizes.append(0)
                    file_checksums.append("Ошибка")
                    file_formats.append("pdf")
                    en_designations.append("")
                    secrecy_grades.append("н/с")
                    notes.append("")
                    a4_counts.append("")
                    a3_counts.append("")
                    a2_counts.append("")
                    a1_counts.append("")
                    a0_counts.append("")
                    additional_counts.append("")
                    a4_sheets_counts.append("")
                    
                    processed_files += 1
                    number_pp.append(processed_files)  # Только один раз!
                    continue
            
            # Закрываем диалоговое окно прогресса
            self.close_progress_dialog()
            
            # Если обработка была отменена
            if self.processing_cancelled:
                self.show_message_box('Отменено', 'Обработка файлов была отменена пользователем.', QMessageBox.Information)
                self.start_button.setEnabled(True)
                self.start_button.setText('Запустить анализ')
                return
            
            # Если все файлы были пропущены из-за ошибок
            if processed_files == 0:
                self.show_message_box('Ошибка', 
                                    f'Не удалось обработать ни один файл.\nВсего файлов: {total_files}\nОшибок: {len(error_files)}', 
                                    QMessageBox.Warning)
                self.start_button.setEnabled(True)
                self.start_button.setText('Запустить анализ')
                return
            
            # После обработки всех файлов, перед созданием DataFrame
            designations_map = {}

            # Список всех известных кодов
            all_known_codes = set()
            all_known_codes.update(self.get_kd_name_mapping().keys())
            for pattern in self.get_pd_name_mapping().keys():
                code = pattern.replace('XXXXX', '')
                if code:
                    all_known_codes.add(code)
            all_known_codes.update(['-ЛУ', '-УЛ', 'ЛУ', 'УЛ', 'НДЗ', 'НД', 'НДЗ-УЛ', 'НД-УЛ', '-01'])

            for i in range(len(file_names)):
                designation = numbers_KD_list[i] if numbers_KD_list[i] else numbers_PD_list[i]
                name_from_file = names[i] if names[i] else ""
                file_name = file_names[i]
                
                if designation:
                    clean_name = name_from_file if name_from_file else ""
                    if not clean_name:
                        temp_name = file_name.replace(designation, '').replace('.pdf', '').strip()
                        temp_name = re.sub(r'[()\[\]_\-]+$', '', temp_name).strip()
                        clean_name = temp_name
                    
                    # Проверка наличия кода документа
                    has_code = False
                    for code in all_known_codes:
                        if not code:
                            continue
                        if f'-{code}' in designation or designation.endswith(code):
                            has_code = True
                            break

                    if re.search(r'[A-ZА-Я]{4}\.\d{5}-\d{2}\s+\d{2}\s+\d{2}', designation):
                        has_code = True
                    
                    if not has_code and clean_name:
                        designations_map[designation] = clean_name
                    elif has_code and clean_name:
                        base_des = self.get_base_designation(designation)
                        if base_des not in designations_map:
                            designations_map[base_des] = clean_name
            self.specification_names_cache.update(designations_map)
            
            # ИЗМЕНЕННЫЙ DataFrame с новым порядком столбцов
            df = pd.DataFrame({
                '№ п/п': number_pp,
                'Наименование документа': ['' if not name else name for name in names],
                'Обозначение документа ТД': numbers_TD_list,
                'Обозначение документа КД': [kd if kd else pd for kd, pd in zip(numbers_KD_list, numbers_PD_list)],  # КД или ПД
                'Обозначение ЭН': en_designations,
                'Обозначение файла КД': insurance_KD,
                'Обозначение файла ТД': insurance_TD,
                'Контрольная сумма': file_checksums,
                'Формат файла': file_formats,
                'Физических листов': page_counts,
                'А4': a4_counts,
                'А3': a3_counts,
                'А2': a2_counts,
                'А1': a1_counts,
                'А0': a0_counts,
                'Доп. форматы': additional_counts,
                'Листов формата А4': a4_sheets_counts,
                'Объем, МБ': file_sizes,
                'Гриф секретности': secrecy_grades,
                'Примечание': notes,
                'Путь к файлу': file_paths,
                'Папка': folders,
                'Имя файла': file_names,
                '_mtime': file_mtimes
            })
            
            # Получаем выбранный тип выгрузки
            export_type = self.export_combo.currentText()
            recursive = self.recursive_checkbox.isChecked()

            # ========== СОРТИРОВКА ==========
            preserve_order = self.preserve_order_checkbox.isChecked()

            if preserve_order:
                # РЕЖИМ ПОРЯДКА ИЗ ПАПКИ: сортируем по дате изменения файла (старые → новые)
                df = df.sort_values('_mtime', ascending=True)
            elif recursive:
                # РЕКУРСИВНЫЙ РЕЖИМ: иерархическая сортировка по папкам и порядку КД
                if export_type in ['Ведомость конструкторской документации', 'Опись конструкторской документации', 'Все данные']:
                    kd_order = self.get_kd_order()

                    def get_sort_key(row):
                        folder = row['Папка']
                        designation = row['Обозначение документа КД']

                        if not designation:
                            return (folder, "", 999, 0, "", 0)

                        # Извлекаем базовый номер (включая -01, -02 и т.д. как часть базового номера)
                        base_number = self.get_base_designation(designation)

                        # Извлекаем номер исполнения для сортировки (если есть)
                        exec_number = 0
                        exec_match = re.search(r'-(\d+)(?=[A-ZА-Я]|$)', base_number)
                        if exec_match:
                            exec_number = int(exec_match.group(1))
                            # Убираем номер исполнения из base_number для группировки
                            base_number = re.sub(r'-\d+', '', base_number)

                        # Очищаем обозначение от суффикса для определения кода
                        clean_for_code = designation
                        has_suffix = 0
                        suffix_type = 0

                        if designation.endswith('-ЛУ'):
                            clean_for_code = designation[:-3]
                            has_suffix = 1
                            suffix_type = 1
                        elif designation.endswith('-УЛ'):
                            clean_for_code = designation[:-3]
                            has_suffix = 1
                            suffix_type = 2

                        # Определяем код документа
                        kd_mapping_local = self.get_kd_name_mapping()
                        all_codes = [c for c in kd_mapping_local.keys() if c not in ['-ЛУ', '-УЛ', 'ЛУ', 'УЛ']]
                        all_codes.sort(key=len, reverse=True)

                        doc_code = None
                        has_code = False

                        for code in all_codes:
                            if f'-{code}' in clean_for_code or clean_for_code.endswith(code):
                                has_code = True
                                doc_code = code
                                break

                        # Определяем порядок
                        if not has_code:
                            # Спецификация - порядок 0
                            order = 0
                            sort_suffix = 0
                        else:
                            # Получаем базовый порядок для кода
                            base_order = kd_order.get(doc_code, 999)
                            if has_suffix:
                                order = base_order + 0.5
                                sort_suffix = suffix_type
                            else:
                                order = base_order
                                sort_suffix = 0

                        # Ключ сортировки: папка, базовый номер (без номера исполнения), номер исполнения, порядок, суффикс
                        return (folder, base_number, exec_number, order, sort_suffix, designation)

                    df['_sort_key'] = df.apply(get_sort_key, axis=1)
                    df = df.sort_values('_sort_key')
                    df = df.drop(columns=['_sort_key'])
                else:
                    # Для ТД - сортировка по папкам и имени файла
                    if 'Папка' in df.columns and 'Имя файла' in df.columns:
                        df = df.sort_values(['Папка', 'Имя файла'])
            else:
                # НЕ РЕКУРСИВНЫЙ РЕЖИМ: простая сортировка по имени файла от А до Я
                if 'Имя файла' in df.columns:
                    df = df.sort_values('Имя файла', ascending=True)

            # Удаляем вспомогательную колонку даты изменения
            df = df.drop(columns=['_mtime'])

            # Обновляем нумерацию после сортировки
            df = df.reset_index(drop=True)
            if '№ п/п' in df.columns:
                df['№ п/п'] = range(1, len(df) + 1)

            # ========== ФИЛЬТРАЦИЯ СТОЛБЦОВ (ДО СОХРАНЕНИЯ!) ==========
            df = self.filter_columns_by_export_type(df, export_type)

            # ЕЩЕ РАЗ ОБНОВЛЯЕМ НУМЕРАЦИЮ ПОСЛЕ ФИЛЬТРАЦИИ
            df = df.reset_index(drop=True)
            if '№ п/п' in df.columns:
                df['№ п/п'] = range(1, len(df) + 1)

            # Проверка, не открыт ли файл в другой программе
            try:
                # Проверяем возможность записи в файл
                if os.path.exists(save_path):
                    with open(save_path, 'a') as f:
                        pass
            except PermissionError:
                error_message = f"Невозможно сохранить файл.\n\nФайл {os.path.basename(save_path)} открыт в другой программе (например, Excel).\n\nЗакройте файл и повторите попытку."
                self.show_message_box('Файл заблокирован', error_message, QMessageBox.Warning)
                self.start_button.setEnabled(True)
                self.start_button.setText('Запустить анализ')
                return

            # Создаем папки, если их нет
            os.makedirs(os.path.dirname(save_path), exist_ok=True)

            # Сохраняем в Excel с именем листа "ОПИСЬ"
            with pd.ExcelWriter(save_path, engine='openpyxl') as writer:
                df.to_excel(writer, sheet_name='ОПИСЬ', index=False)

            # Применяем форматирование к Excel файлу
            self.format_excel_file(save_path)

            # Формируем текст статистики (используем уже отфильтрованный df)
            stats_text = (
                f"Отчет успешно создан!\n\n"
                f"Сохранен в: {save_path}\n\n"
                f"Статистика:\n"
                f"- Всего найдено файлов: {total_files}\n"
                f"- Успешно обработано: {processed_files}\n"
            )

            # Добавляем статистику в зависимости от типа выгрузки
            if export_type == 'Ведомость технологической документации':
                stats_text += f"- Файлов с обозначением ТД: {df['Обозначение документа ТД'].astype(bool).sum()}\n"
            elif export_type == 'Ведомость конструкторской документации':
                stats_text += f"- Файлов с обозначением КД/ПД: {df['Обозначение документа КД'].astype(bool).sum()}\n"
            elif export_type == 'Опись конструкторской документации':
                stats_text += f"- Файлов в описи КД/ПД: {len(df)}\n"
            else:  # 'Все данные'
                stats_text += f"- Файлов с обозначением КД/ПД: {df['Обозначение документа КД'].astype(bool).sum()}\n"
                stats_text += f"- Файлов с обозначением ТД: {df['Обозначение документа ТД'].astype(bool).sum()}\n"

            # Добавляем описание типа выгрузки
            if export_type == 'Ведомость технологической документации':
                stats_text += f"Выгружен отчет по технологической документации (ТД).\n"
            elif export_type == 'Ведомость конструкторской документации':
                stats_text += f"Выгружен отчет по конструкторской/программной документации (КД/ПД).\n"
            elif export_type == 'Опись конструкторской документации':
                stats_text += f"Выгружена опись конструкторской/программной документации (КД/ПД) - сокращенный формат.\n"
            else:
                stats_text += f"Выгружен полный отчет со всеми данными.\n"

            # Если были ошибки, добавляем информацию о них
            if error_files:
                stats_text += f"\n\nФайлы с ошибками ({len(error_files)}):\n"
                for error in error_files[:5]:  # Показываем только первые 5 ошибок
                    stats_text += f"- {error}\n"
                if len(error_files) > 5:
                    stats_text += f"... и еще {len(error_files) - 5} файлов с ошибками\n"

            # Показываем диалоговое окно со статистикой
            self.show_stats_dialog(stats_text, save_path)
            
        except PermissionError as e:
            # Специальная обработка ошибки доступа к файлу
            error_message = f"Ошибка доступа к файлу.\n\nВозможно, файл отчета открыт в другой программе (например, Excel).\n\nЗакройте файл и повторите попытку.\n\nПуть к файлу: {save_path}"
            self.show_message_box('Ошибка доступа к файлу', error_message, QMessageBox.Critical)
            
        except Exception as e:
            # Закрываем диалоговое окно прогресса если оно открыто
            if hasattr(self, 'progress_dialog') and self.progress_dialog:
                self.close_progress_dialog()
            
            self.show_message_box('Ошибка', f'Произошла ошибка: {str(e)}', QMessageBox.Critical)
        finally:
            # Включаем кнопку обратно
            self.start_button.setEnabled(True)
            self.start_button.setText('Запустить анализ')
            self.processing_cancelled = False  # Сбрасываем флаг отмены
