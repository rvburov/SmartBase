//   2. Переименовать файл и класс: «MyAutomation» → по смыслу, латиницей.
//   3. Вписать Key, Name, What и логику в Run.
//   4. Собрать. Все — новый пункт уже стоит в списке шагов сценария: реестр находит
//      автоматизации по сборке, править ни его, ни окно настройки шага не нужно.
//
// Расширение файла — .cs.txt намеренно: заготовка не собирается и не мешает сборке, пока ее
// не скопировали. У копии расширение обычное — .cs.

namespace SmartBase.Extensions.Automations;

/// <summary>
/// Что делает автоматизация — одним абзацем. Это описание видно человеку в списке шагов, так
/// что писать его стоит на языке того, кто настраивает обмен, а не того, кто пишет код.
/// </summary>
public sealed class MyAutomation : IAutomation
{
    /// <summary>Ключ хранится в шаге сценария: слаг латиницей, стабильный между версиями.</summary>
    public string Key => "my-automation";

    /// <summary>Имя в списке шагов.</summary>
    public string Name => "Моя автоматизация";

    /// <summary>Одна строка под именем — что она делает.</summary>
    public string What => "Считает то-то по строкам предыдущего шага.";

    /// <summary>
    /// Настройки шага: их человек заполняет один раз, когда ставит шаг. Подстановок
    /// {{колонка}} в них нет — поле одно, а строк много. Настроек не нужно — уберите свойство.
    /// </summary>
    public IReadOnlyList<ExtensionActionField> Settings =>
    [
        new("column", "Колонка", Watermark: "Проект"),
    ];

    /// <summary>
    /// Строки пришли — строки ушли. Все, что доступно: input.Rows и input.Setting(…).
    /// Ни базы, ни сети, ни файлов у автоматизации нет — и не будет.
    ///
    /// Пустой список означает «дальше идти незачем»: сценарий кончится на этом шаге без жалобы.
    /// Отказ — throw new AutomationRefusal("…"): текст доедет до отчета о запуске и до
    /// квитанции отправителю как есть.
    /// </summary>
    public IReadOnlyList<AutomationRow> Run(AutomationInput input)
    {
        var column = input.Setting("column");

        if (column.Length == 0) throw new AutomationRefusal("не названа колонка");

        var made = new List<AutomationRow>(input.Rows.Count);

        foreach (var row in input.Rows)
        {
            // row["Имя"]         — значение колонки (регистр не важен, нет колонки — пусто);
            // row.Has("Имя")     — есть ли такая колонка вообще;
            // row.Number("Цена") — число или null, если там не число;
            // row.With("Итого", 120)  — КОПИЯ строки с новой колонкой.
            var value = row.Number(column) ?? 0;

            made.Add(row.With("Итого", value * 2));
        }

        return made;
    }
}
