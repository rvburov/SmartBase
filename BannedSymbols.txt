# Запрещенные вызовы — второй рубеж правила «наружу одна дверь»
# (docs/ROADMAP-API.md, 10.2; docs/developer/features/integrations/overview.md).
# Первый рубеж — текстовый тест Tests/Unit/OneDoorOutwardTests.cs: он читает исходники и
# ловит даже упоминание сетевого типа в комментарии. Этот список ловит другое: тип,
# использованный БЕЗ упоминания имени — через var, псевдоним using, наследника или
# обобщенный параметр. Текстом такое не найти, а компилятору видно.
# Снимается запрет строкой «#pragma warning disable RS0030», и мест с ней в репозитории три:
# дверь наружу (Integrations/EgressHttp.Open.cs), дверь внутрь (Integrations/IngressHttp.Open.cs)
# и проба порта PostgreSQL (Utils/PgServerProbe.cs — своя база приложения, а не чужая программа).
# Четвертое роняет тест OneDoorOutwardTests.
# Формат: «ID_символа;сообщение». Пустых строк и строк из одного знака «#» тут быть не может:
# анализатор считает их записями без символа и ругается RS0031 — поэтому текст сплошной.
# ── Клиенты HTTP ──────────────────────────────────────────────────────────────
T:System.Net.Http.HttpClient;наружу ходит только Integrations/EgressHttp.Open.cs
T:System.Net.Http.HttpClientHandler;наружу ходит только Integrations/EgressHttp.Open.cs
T:System.Net.Http.SocketsHttpHandler;наружу ходит только Integrations/EgressHttp.Open.cs
T:System.Net.Http.HttpMessageInvoker;наружу ходит только Integrations/EgressHttp.Open.cs
T:System.Net.Http.HttpRequestMessage;запрос наружу собирает EgressPlan в Integrations/EgressCall.cs
T:System.Net.WebClient;устаревший клиент; наружу ходит только Integrations/EgressHttp.Open.cs
T:System.Net.WebRequest;устаревший клиент; наружу ходит только Integrations/EgressHttp.Open.cs
T:System.Net.HttpWebRequest;устаревший клиент; наружу ходит только Integrations/EgressHttp.Open.cs
T:System.Net.WebSockets.ClientWebSocket;поток событий открывает только Integrations/EgressHttp.Open.cs
# ── Прием входящих по сети: слушает одна дверь ────────────────────────────────
# Слушатель у программы один — Integrations/IngressHttp.Open.cs, петлевой адрес по умолчанию
# и выключенный прием в состоянии по умолчанию (10.12). Второй слушатель где угодно еще — это
# второй набор правил о том, кого пускать, и узнать о нем можно было бы только чтением кода.
# Вебхуки из интернета по-прежнему названы в разделе 13 среди того, чего не будет: наружу
# программа не выставляется, вместо этого опрос или релей.
T:System.Net.HttpListener;слушает только Integrations/IngressHttp.Open.cs
T:System.Net.Sockets.TcpListener;слушает только Integrations/IngressHttp.Open.cs

# Разрешение имени — уже обращение наружу: сам запрос к DNS говорит чужой сети, что программа
# этим адресом интересуется (10.5). Поэтому Dns здесь наравне с сокетом.
T:System.Net.Sockets.Socket;сокет открывает только Integrations/EgressHttp.Open.cs
T:System.Net.Sockets.TcpClient;сокет открывает только Integrations/EgressHttp.Open.cs
T:System.Net.Sockets.UdpClient;сокет открывает только Integrations/EgressHttp.Open.cs
